    export const CreateJobSchema = z.object({
      // Required fields (D-11)
      industry: z.string().min(1, '行业不能为空'),

    **Step 2: Modify `src/app/api/jobs/route.ts`**

    Replace the current body validation with Zod:

    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { randomUUID } from 'crypto'
    import * as db from '@/lib/db'
    import { CreateJobSchema } from '@/lib/schemas'

    export async function POST(req: NextRequest) {
      try {
        const rawBody = await req.json()
      } catch {
        return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 })
      }

      const rawBody = await req.json()
      const result = CreateJobSchema.safeParse(rawBody)

      if (!result.success) {
        return NextResponse.json(
          { error: 'Validation failed', details: result.error.issues },
          { status: 400 }
        )
      }

      const data = result.data
      const mode = data.mode
      const storeProfile = { ...data }
      delete (storeProfile as any).mode

      const id = randomUUID()
      db.createJob(id, JSON.stringify(storeProfile), mode)
      return NextResponse.json({ id, runStatus: 'queued' }, { status: 201 })
    }
    ```

    Important: The schema validates a flat object. After parsing, extract mode, then pass the rest as storeProfile JSON. This avoids the old nested `body.storeProfile` structure -- the schema defines the full flat shape.

    Note on `safeParse` vs `parse`: Use `safeParse` so we control the error response format (400 with field details), rather than letting Zod throw and relying on Next.js error handling.
